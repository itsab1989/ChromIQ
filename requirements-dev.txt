# Development / test-only dependencies.
#
# Install the runtime deps first (see requirements.txt), then:
#     pip install -r requirements-dev.txt
#
# Kept separate from requirements.txt on purpose: requirements.txt pins
# pycups, which only builds on macOS/Linux. On Windows install the runtime
# deps with pycups filtered out (as the build-windows workflow does), then
# add these test tools.
#
# Run the suite:  QT_QPA_PLATFORM=offscreen pytest   (see CLAUDE.md)

pytest>=8.0
pytest-qt>=4.4      # provides the `qapp` fixture used across the UI tests
pytest-xdist>=3.8   # parallel runs: -n 8 --dist loadfile. loadfile keeps a
                    # file's tests in one worker — order within a file is
                    # load-bearing in places, and this suite has had a
                    # stylesheet leak travel 400 tests before now.
pypdf>=4.0          # test-only: parses the vector-PDF export for fidelity checks
pymupdf                # scripts/make_workflow_diagram.py (diagram generation only)
